% Fig. 7 — every instrumented guided tour, in time order.
% Used by sections/06-evaluation.
%
% This figure replaces a paragraph. It has to show three things at once, which
% prose does badly: when each tour ran, how far it got, and how it ended. The
% vertical rule is the point where the map and its annotations stopped
% changing, and it is the whole argument: no navigation failure occurs to its
% right. Marker shape carries the outcome so the figure survives greyscale.
%
% Data: data/logs/tour/*.log, 24 sessions, day 0 = 23 June 2026.
\begin{figure}[!t]
\centering
\begin{tikzpicture}
\begin{axis}[
  width=\columnwidth, height=5.0cm,
  xlabel={Days from first instrumented tour}, ylabel={Legs completed},
  label style={font=\scriptsize}, tick label style={font=\scriptsize},
  xmin=-2, xmax=53, ymin=-0.6, ymax=10.4,
  ytick={0,3,6,9}, xtick={0,10,20,30,40,50},
  grid=major, grid style={gray!20},
  legend style={font=\tiny, at={(0.5,-0.32)}, anchor=north,
                legend columns=3, draw=none, column sep=4pt},
]
  % frontiere : carte et annotations stabilisees
  \draw[dashed, thick, draw=figdark] (axis cs:2.95,-0.6) -- (axis cs:2.95,10.4);
  \node[font=\tiny, text=figdark, anchor=west, align=left]
    at (axis cs:5.5,5.4) {map and annotations stable};

  \addplot[only marks, mark=x, mark size=2.6pt, draw=red!70!black, thick]
    coordinates {(0,0) (0,0) (1,0) (2,1) (2,0) (2,0) (3,2) (50,0) (50,0)};
  \addlegendentry{aborted}

  \addplot[only marks, mark=o, mark size=2.4pt, draw=figdark]
    coordinates {(2,0) (6,0) (23,1) (23,0) (30,8) (30,3) (30,2) (30,1)};
  \addlegendentry{stopped by operator}

  \addplot[only marks, mark=*, mark size=2.4pt, draw=green!45!black, fill=green!55!black]
    coordinates {(3,9) (5,9) (6,9) (29,9) (29,9) (29,9) (29,9)};
  \addlegendentry{ran to completion}
\end{axis}
\end{tikzpicture}
\caption{All \num{24} instrumented guided tours. To the left of the rule, while the map and its
annotations were still being edited, one navigation leg in seven succeeded and no tour finished.
To the right, \num{80} consecutive legs succeeded with no navigation failure, and seven tours ran
end to end. Operator stops are not failures: every leg they had attempted had succeeded. The two
late aborts ended on a speech failure, not a navigation one.}
\label{fig:tourtimeline}
\end{figure}
